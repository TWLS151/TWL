```python
# 정사각형 방
# 처음에 문제를 보고 완전탐색으로 풀까? 생각을 했는데
# 좌표 간 차이로 연속하는지 안하는지 확인할 수 있을 것이라 생각함
import sys
sys.stdin = open('1861.txt')

T = int(input())
for tc in range(1, T+1):
    N = int(input())
    arr = [list(map(int, input().split())) for _ in range(N)]
    
    # pos[x] = 숫자 x가 있는 (r, c)의 좌표
    pos = [0] * ((N*N)+1)
    
    # 모든 방을 돌면서 숫자-좌표를 매핑
    for r in range(N):
        for c in range(N):
            # ex) if arr[5][3] = 10:
            #       pos[10] == (5, 3)
            pos[arr[r][c]] = (r, c)
    
    best_start = 1  # 가장 많이 이동할 수 있는 방의 수를 가진 시작점
    max_val = 1     # 가장 많이 이동할 수 있는 방의 수

    c_start = 1     # 현재 구간의 시작 숫자
    c_val = 1       # 현재 이동할 수 있는 방의 수
    
    # 숫자를 1부터 N^2-1까지 탐색
    for i in range(1, N*N):
        r1, c1 = pos[i]
        r2, c2 = pos[i+1]
        
        # 두 방이 상하좌우 중 하나로 인접한 경우
        if abs(r1-r2) + abs(c1-c2) == 1:
            c_val += 1
            
        # 이동이 끊긴 경우
        # 지금까지의 이동 방의 수와 최대 방의 수 갱신
        else:
            if c_val > max_val:
                max_val = c_val
                best_start = c_start
                
            # 길이가 같다면 시작점 숫자가 작은 것 선택
            elif c_val == max_val:
                best_start = min(best_start, c_start)
            
            # 다음 반복 구간 시작
            c_start = i+1
            c_val = 1
    
    # 반복문이 끝났을 때 마지막 구간 처리
    # 혹시 마지막 숫자까지 연속이 이어졌을 수도 있기에
    if c_val > max_val:
        max_val = c_val
        best_start = c_start
    elif c_val == max_val:
        best_start = min(best_start, c_start)

    print(f'#{tc} {best_start} {max_val}')
```

# 격자판의 숫자 이어붙이기
# 주석을 쓰다보니까 그냥 arr를 받을 때 int말고
# 그냥 str로 받았어야했다.
# 그게 훨씬 나을듯

import sys
sys.stdin = open('2819.txt')

# 델타
dr = [0, 1, 0, -1]
dc = [1, 0, -1, 0]

T = int(input())
N = 4
for tc in range(1, T+1):
    arr = [list(map(int, input().split())) for _ in range(N)]
    
    # 서로 다른 7자리 수릐 개수를 구해야하므로
    # 중복을 거르기 위한 set
    # 최종적으로 len(ans)를 제출할 예정
    ans = set()

    stack = []
    
    # arr을 순회하며 스택에 좌표 r, c값과 해당 값을 push
    for r in range(N):
        for c in range(N):
            stack.append((r, c, str(arr[r][c])))
    
    # 스택이 빌 동안
    while stack:
        # (열 좌표, 행 좌표, 쌓인 숫자)
        r, c, s = stack.pop()
        
        # 숫자가 7자리가 되면 ans세트에 추가
        if len(s) == 7:
            ans.add(s)
            continue
        
        # 상하좌우 방향에 해당하는 좌표들을
        # 스택에 담음
        for d in range(4):
            nr, nc = r+dr[d], c+dc[d]
            if 0 <= nr < N and 0 <= nc < N:
                stack.append((nr, nc, s+str(arr[nr][nc])))

    print(f'#{tc} {len(ans)}')

# 햄버거 다이어트
# 민기의 문제는 햄버거가 아닌듯
# 재귀로 구현하기보다 스택으로 구현하는게 편한듯...

T = int(input())
for tc in range(1, T+1):
    N, L = map(int, input().split())
    info = [list(map(int, input().split())) for _ in range(N)]

    # 최대 맛 점수
    max_val = 0
    
    # 스택의 요소
    # (재료의 인덱스, 누적 맛 점수, 누적 칼로리)
    stack = [(0, 0, 0)]

    while stack:
        idx, val, cost = stack.pop()
        
        # 누적된 칼로리가 L보다 크면 pass
        if cost > L:
            continue
        
        # 재료를 끝까지 골랐으면
        # 최댓값 갱신
        if idx == N:
            if val > max_val:
                max_val = val
            continue
        
        # 해당 인덱스를 고르지 않음
        stack.append((idx+1, val, cost))
        
        # 해당 인덱스를 고름
        stack.append((idx+1, val+info[idx][0], cost + info[idx][1]))

    print(f'#{tc} {max_val}')
