# 1027 고층 건물
# 완전탐색으로 기준점과 다음 빌딩의 기울기를 잰 후 리스트에 넣음
# 리스트에서 마지막 값과 다음 빌딩 기울기를 비교해서
# 볼 수 있는 건물을 카운트
# 밑에 있는 주석 코드는 gpt의 최적화

N = int(input())
arr = list(map(int, input().split()))

max_cnt = 0

for i in range(N):
    cnt = 0
    if i != N-1:
        r = 1
        slopes = []
        while i+r-1 != N-1:
            slope = (arr[i+r] - arr[i]) / r
            if not slopes:
                slopes.append(slope)
                r += 1
                cnt += 1
            else:
                if slopes[-1] < slope:
                    cnt += 1
                    r += 1
                    slopes.append(slope)
                else:
                    r += 1
                    continue

    if i != 0:
        l = 1
        slopes = []
        while i-l+1 != 0:
            slope = (arr[i-l] - arr[i]) / l
            if not slopes:
                slopes.append(slope)
                l += 1
                cnt += 1
            else:
                if slopes[-1] < slope:
                    cnt += 1
                    l += 1
                    slopes.append(slope)
                else:
                    l += 1
                    continue

    if max_cnt < cnt:
        max_cnt = max(max_cnt, cnt)

print(max_cnt)

# 4485 녹색 옷 입은 애가 젤다지?
# 다익스트라
# 무난한 2차원 다익스트라 문제인 것 같음
# 근데 SWEA랑 입출력이 좀 달라서 당황한 것 빼면 쉬운 문제

import heapq

dr = [0, 1, 0, -1]
dc = [1, 0, -1, 0]
tc = 1
while True:
    N = int(input())
    if N == 0:
        break
    arr = [list(map(int, input().split())) for _ in range(N)]

    costs = [[float('inf')] * N for _ in range(N)]

    pq = []
    heapq.heappush(pq, (arr[0][0], 0, 0))
    costs[0][0] = arr[0][0]

    while pq:
        cost, r, c = heapq.heappop(pq)

        if cost > costs[r][c]:
            continue

        for d in range(4):
            nr, nc = r+dr[d], c+dc[d]
            if 0 <= nr < N and 0 <= nc < N:
                ncost = cost + arr[nr][nc]

                if ncost < costs[nr][nc]:
                    costs[nr][nc] = ncost
                    heapq.heappush(pq, (ncost, nr, nc))

    ans = costs[N-1][N-1]
    print(f'Problem {tc}: {ans}')
    tc += 1

# 9012 괄호
# 예전에 이런 문제 어떻게 풀었는 지 기억이 안나서 그냥 되는대로 풀었음

T = int(input())
for _ in range(T):
    lst = list(input())

    stack = []
    flag = False
    for x in lst:
        if x == '(':
            stack.append(1)
        elif x == ')':
            if stack:
                stack.pop()
            else:
                print('NO')
                flag = True
                break
    if not flag:
        if stack:
            print('NO')
        else:
            print('YES')
